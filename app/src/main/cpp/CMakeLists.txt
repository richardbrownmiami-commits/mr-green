cmake_minimum_required(VERSION 3.22.1)
project(mrgreen LANGUAGES C CXX)
if(NOT ANDROID OR NOT ANDROID_ABI STREQUAL "armeabi-v7a")
    message(FATAL_ERROR "Mr Green native engine is supported only on Android armeabi-v7a")
endif()
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_OPENCL OFF CACHE BOOL "" FORCE)
set(GGML_SYCL OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
set(GGML_RPC OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ALL_VARIANTS OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ARM_ARCH "armv7-a" CACHE STRING "" FORCE)
# Debug otherwise compiles the entire inference engine without optimization.
# Keep debugging and assertions, but optimize every native dependency too.
add_compile_options("$<$<CONFIG:Debug>:-O3>" -mfpu=neon -mfloat-abi=softfp)
set(LLAMA_VENDOR_DIR "${CMAKE_CURRENT_LIST_DIR}/../../../../vendor/llama.cpp")
if(NOT EXISTS "${LLAMA_VENDOR_DIR}/CMakeLists.txt")
    message(FATAL_ERROR "Pinned llama.cpp missing. Run scripts/fetch-llama.sh")
endif()
add_subdirectory("${LLAMA_VENDOR_DIR}" llama.cpp)
add_library(mrgreen SHARED llama_jni.cpp)
target_compile_features(mrgreen PRIVATE cxx_std_17)
target_include_directories(mrgreen PRIVATE "${LLAMA_VENDOR_DIR}/include")
target_link_libraries(mrgreen PRIVATE llama log android)
# Generic ARMv7 NEON only; no host-native tuning, dotprod or fp16-arith.
